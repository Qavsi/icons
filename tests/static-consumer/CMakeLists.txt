cmake_minimum_required(VERSION 3.30)
project(StaticConsumer LANGUAGES CXX)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick)
qt_standard_project_setup(REQUIRES 6.8)
include(FetchContent)
block(SCOPE_FOR VARIABLES)
    set(BUILD_SHARED_LIBS OFF)
    set(QAVSI_ICONS_BUILD_TESTS OFF)
    FetchContent_Declare(qavsi_icons SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../..")
    FetchContent_MakeAvailable(qavsi_icons)
endblock()
qt_add_executable(static-consumer main.cpp)
target_link_libraries(static-consumer PRIVATE Qt6::Quick QavsiIconsCorePlugin QavsiIconsMaterialPlugin)
enable_testing()
add_test(NAME static-consumer COMMAND static-consumer)
set_tests_properties(static-consumer PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
